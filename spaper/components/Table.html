<script>
const data = [
  { rank: 1, name: 'Jannik Sinner', country: 'Italy', countryCode: 'it' },
  { rank: 2, name: 'Alexander Zverev', country: 'Germany', countryCode: 'de' },
  { rank: 3, name: 'Carlos Alcaraz', country: 'Spain', countryCode: 'es' },
  { rank: 4, name: 'Ben Shelton', country: 'USA', countryCode: 'us' },
];
</script>

<Table {data}>
  {#snippet children({ row })}
    <TableColumn field="rank" label="Rank" {row}>
      {#snippet children({ value })}
        <Tag type={value === 1 ? 'success' : 'primary'}>{value}</Tag>
      {/snippet}
    </TableColumn>
    <TableColumn field="name" label="Name" {row} />
    <TableColumn field="countryCode" label="Country" {row}>
      {#snippet children({ row, value })}
        <span class="inline-flex items-center gap-2">
          <img alt="" class="border-paper" src="https://flagcdn.com/w40/{value}.png" />
          {row.country}
        </span>
      {/snippet}
    </TableColumn>
  {/snippet}
</Table>

API

- Table

NameDescriptionTypeValuesDefault
dataTable dataArray_[]
columnsColumn fields or { field, label, sortable } objects. `content` and `html` are deprecated; use TableColumnArray_[]
stripedStriped styleBoolean_false
hoverableHighlight line on hoverBoolean_false
sortableDefault for columns that do not set their own. Header clicks cycle the sortBoolean_false
sortActive sort. order 1 is ascending, -1 is descending. A header click cycles ascending, descending, then unsorted.{ field: string, order: 1 | -1 } | null_null
showHeaderWether to show header or notBoolean_true
headerCLassCSS classes to be applied on headerString__
cellCLassCSS classes to be applied on cellString__
childrenRow content. Receives { row, rowIndex }Snippet__
emptyShown when data is emptySnippet__
NameDescriptionDefault
--paper-table-header-hoverBackground of a sortable column header on hovervar(--paper-table-hover-bg)
